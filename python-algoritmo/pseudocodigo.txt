algoritmo "tabuada_do_9"

var
    n         : inteiro
    dezena    : inteiro
    unidade   : inteiro
    resultado : inteiro

inicio
    // ENTRADA
    repita
        escreva("Digite um número inteiro de 1 a 10: ")
        leia(n)
        se (n < 1) ou (n > 10) entao
            escreva("Valor inválido! Tente novamente.")
        fimse
    ate (n >= 1) e (n <= 10)

    // PROCESSAMENTO
    dezena    <- n - 1
    unidade   <- 10 - n
    resultado <- dezena * 10 + unidade

    // SAÍDA
    escreva("9 x ", n, " = ", resultado)
fimalgoritmo
